    toLocation: string;
    reason: string;

    odometerEnd: number | null;
  };

export interface ImportPreviewResult {

Create `src/modules/logbook/import/parse-dates.ts` with three exported functions:

1. `parseSaDateString(raw: string): string | null` — DD/MM/YYYY-first parsing using the already-installed date-fns: try `parse(trimmed, "dd/MM/yyyy", new Date())`, then `"d/M/yyyy"`, then accept an already-ISO `yyyy-MM-dd` string (validate it round-trips). Return ISO `YYYY-MM-DD` or null. Reject calendar-impossible dates (31/02/2026 -> null) via date-fns `isValid`. NEVER use `new Date(raw)` on the input string — that is the locale-ambiguity bug this phase exists to fix.
2. `excelSerialDateToIso(serial: number): string | null` — `Math.round((serial - 25569) * 86400 * 1000)` epoch conversion (Excel epoch 1899-12-30), return `toISOString().slice(0, 10)`. Return null for non-finite input or serials outside a plausible range (roughly 20000–80000, i.e. ~1954–2118) so arbitrary numbers are not silently treated as dates.
3. `normalizeDateCell(value: unknown): string | null` — the single dispatcher both parsers/mapping use: `Date` instance -> ISO date string; string matching ISO -> validated passthrough; other string -> parseSaDateString; number (or numeric string) in serial range -> excelSerialDateToIso; anything else -> null.

Write `parse-dates.test.ts` (Vitest, co-located) covering at minimum:
- "05/06/2026" -> "2026-06-05" (DD/MM-first, the ambiguous-date case)
- "3/4/2026" -> "2026-04-03" (single-digit day/month)
- "31/02/2026" -> null (calendar-impossible)
- "2026-06-05" -> "2026-06-05" (ISO passthrough)
- excelSerialDateToIso(45658) -> "2025-01-01" and 45292 -> "2024-01-01" (known reference serials)
- excelSerialDateToIso(123) -> null (out of plausible range)
- normalizeDateCell dispatch: Date instance, "05/06/2026", 45658, "45658", "garbage" -> null, "" -> null
