```typescript
import type { TravelDeemedCostBracket } from "@/modules/individual-tax/types";
import type { ActualCostExpenses, LogbookCostMethod, LogbookTravelResult, LogbookWarning } from "@/modules/logbook/types";

/** s8(1)(b)(iiiA)(bb) statutory vehicle-value cap for wear-and-tear and finance-charge purposes.
 *  MEDIUM confidence — verified via two independent secondary sources (taxconsulting.co.za,
 *  Tax Faculty FAQ #1117) citing the same figure and section. TODO(compliance-review):
 *  confirm against SARS IN47 (Issue 5) / current PAYE-GEN-01-G03 PDF before first real client filing. */
export const ACTUAL_COST_VEHICLE_VALUE_CAP = 665_000;
const WEAR_AND_TEAR_WRITE_OFF_YEARS = 7;

function r2(value: number) { return Math.round(value * 100) / 100; }
```

- `calculateDeemedCost(vehicleCostPrice: number, businessKm: number, totalKm: number | null, table: TravelDeemedCostBracket[]): number` — find bracket where `costPrice >= min && (max === null || costPrice <= max)`, falling back to the last bracket. **CRITICAL — SARS method (deviation from the research doc's example, which omitted this step):** the fixed cost is APPORTIONED over total kilometres travelled in the year, not claimed in full: `denominator = totalKm && totalKm > 0 ? Math.max(totalKm, businessKm) : businessKm`; `ratePerKm = bracket.fixedCostAnnual / denominator + bracket.fuelCostPerKm + bracket.maintenanceCostPerKm`; return `r2(ratePerKm * businessKm)`. Return 0 when businessKm <= 0. Add a doc comment citing PAYE-GEN-01-G03-A01 ("fixed cost divided by total kilometres travelled in the year"; full-year vehicle use assumed — part-year pro-rata by days is a documented non-goal this phase).
- `calculateActualCost(vehicleCostPrice: number, businessKm: number, totalKm: number | null, expenses: ActualCostExpenses): number` — `cappedValue = Math.min(vehicleCostPrice, ACTUAL_COST_VEHICLE_VALUE_CAP)`; `wearAndTear = cappedValue / WEAR_AND_TEAR_WRITE_OFF_YEARS`; finance charges: include `expenses.financeCharges` scaled by `cappedValue / vehicleCostPrice` when costPrice exceeds the cap (the defensible pro-rata interpretation — comment it as `TODO(compliance-review)` per research Open Question 2; exact statutory mechanics underdocumented, output is advisory); `total = fuel + maintenance + insurance + licence + cappedFinanceCharges + wearAndTear`; `ratio = min(1, businessKm / denominator)` using the same denominator convention as deemed; return `r2(total * ratio)`.
- `buildTravelResult(input: { businessKm: number; totalKm: number | null; costMethod: LogbookCostMethod; vehicleCostPrice: number; actualExpenses: ActualCostExpenses | null }, table: TravelDeemedCostBracket[]): LogbookTravelResult` — computes BOTH methods independently (`deemedCostDeduction` always; `actualCostDeduction` only when `actualExpenses !== null`, else null + warning `ACTUAL_EXPENSES_MISSING`); `claimedDeduction` resolved by an explicit `if (costMethod === "DEEMED") ... else ...` switch — NEVER `??`/fallback-chain on data presence (Pitfall 1's exact warning sign); ACTUAL elected with null expenses claims 0 with the warning; `recommendedMethod` = method with higher computable deduction (DEEMED when actual is null); warning `MISSING_TOTAL_KM` when totalKm is null or <= 0.

Do NOT hardcode any rate table (Anti-Pattern: rulepack duplication). Do not round anywhere except the final `r2` per function.
