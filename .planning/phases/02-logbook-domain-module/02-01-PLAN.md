```typescript
export type LogbookCostMethod = "DEEMED" | "ACTUAL";

export interface VehicleDetails {
  id: string;
  make: string;
  model: string;
  registrationNumber: string;
  /** Purchase/cost price in rand, incl. VAT, excl. finance charges (SARS s8(1)(b) valuation basis). */
  costPrice: number;
  /** Optional ISO date (YYYY-MM-DD). */
  acquisitionDate?: string | null;
}

export interface LogbookTripRecord {
  id: string;
  /** ISO date YYYY-MM-DD. */
  date: string;
  businessKm: number;
  fromLocation: string;
  toLocation: string;
  reason: string;
  /** Per-trip odometer readings are OPTIONAL per the official SARS eLogbook ("not compulsory"). */
  odometerStart?: number | null;
  odometerEnd?: number | null;
  createdAt: string;
  updatedAt: string;
}

export interface ActualCostExpenses {
  fuel: number;
  maintenance: number;
  insurance: number;
  licence: number;
  financeCharges: number;
}

export interface LogbookRecord {
  id: string;
  clientId: string;
  /** SARS year of assessment (2025–2027 this milestone). Set at creation; never re-derived from trip dates. */
  assessmentYear: number;
  vehicle: VehicleDetails;
  openingOdometer: number;
  closingOdometer: number | null;
  /** Elected once per logbook (per vehicle per tax year) — NEVER per trip (Pitfall 1). */
  costMethod: LogbookCostMethod;
  /** May be captured regardless of elected method — needed for the LOG-05 side-by-side comparison.
      The elected costMethod alone decides the claim; presence of expenses never overrides it. */
  actualExpenses: ActualCostExpenses | null;
  trips: LogbookTripRecord[];
  createdAt: string;
  updatedAt: string;
}

export interface LogbookWarning {
  code: string;
  message: string;
}

/** The decoupled output seam Phase 3 (ITR12 travel schedule) will consume. */
export interface LogbookTravelResult {
  totalKilometres: number | null;   // closing - opening; null when closing odometer not yet recorded
  businessKilometres: number;
  costMethod: LogbookCostMethod;
  deemedCostDeduction: number;
  actualCostDeduction: number | null; // null if actual-cost inputs incomplete
  /** Resolved SOLELY via costMethod — never a data-presence fallback chain. */
  claimedDeduction: number;
  recommendedMethod: LogbookCostMethod; // whichever yields the higher deduction (of the computable ones)
  warnings: LogbookWarning[];
}

/** Data shape for the SARS-audit printable summary (rendered by Phase 6 UI; data-only here per LOG-06). */
export interface LogbookAuditSummary {
  clientId: string;
  assessmentYear: number;
  vehicle: VehicleDetails;
  openingOdometer: number;
  closingOdometer: number | null;
  totalKilometres: number | null;
  totalBusinessKm: number;
  tripCount: number;
  trips: LogbookTripRecord[];
  travelResult: LogbookTravelResult;
  generatedAt: string;
}
```

Follow existing conventions: named exports only, no barrel file, doc comments on non-obvious fields (matches Phase 1's `TravelDeemedCostBracket` style). Dates are ISO strings (not Date objects) because demo-mode records round-trip through JSON — same convention as `IndividualTaxAssessmentRecord.assessmentDate`.

```prisma
enum LogbookCostMethod {
  DEEMED
  ACTUAL
}

model Vehicle {
  id                 String    @id @default(cuid())
  clientId           String
  make               String
  model              String
  registrationNumber String
  costPrice          Decimal   @db.Decimal(18, 2)
  acquisitionDate    DateTime?
  createdAt          DateTime  @default(now())
  updatedAt          DateTime  @updatedAt
  client             Client    @relation(fields: [clientId], references: [id], onDelete: Cascade)
  logbooks           Logbook[]

  @@index([clientId])
}

model Logbook {
  id                   String            @id @default(cuid())
  clientId             String
  vehicleId            String
  assessmentYear       Int
  openingOdometer      Decimal           @db.Decimal(10, 1)
  closingOdometer      Decimal?          @db.Decimal(10, 1)
  costMethod           LogbookCostMethod @default(DEEMED)
  actualFuel           Decimal?          @db.Decimal(18, 2)
  actualMaintenance    Decimal?          @db.Decimal(18, 2)
  actualInsurance      Decimal?          @db.Decimal(18, 2)
  actualLicence        Decimal?          @db.Decimal(18, 2)
  actualFinanceCharges Decimal?          @db.Decimal(18, 2)
  createdAt            DateTime          @default(now())
  updatedAt            DateTime          @updatedAt
  client               Client            @relation(fields: [clientId], references: [id], onDelete: Cascade)
  vehicle              Vehicle           @relation(fields: [vehicleId], references: [id], onDelete: Cascade)
  trips                LogbookTrip[]

  @@unique([clientId, vehicleId, assessmentYear])
  @@index([clientId, assessmentYear])
}

model LogbookTrip {
  id            String   @id @default(cuid())
  logbookId     String
  date          DateTime
  odometerStart Decimal? @db.Decimal(10, 1)
  odometerEnd   Decimal? @db.Decimal(10, 1)
  businessKm    Decimal  @db.Decimal(10, 1)
  fromLocation  String
  toLocation    String
  reason        String
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
  logbook       Logbook  @relation(fields: [logbookId], references: [id], onDelete: Cascade)

  @@index([logbookId, date])
}
```

Also add the two back-relations to the existing `Client` model body: `vehicles Vehicle[]` and `logbooks Logbook[]` (Prisma requires both sides of a relation).

Per-trip odometer fields are nullable BY DESIGN (official SARS eLogbook marks them "not compulsory"); `costMethod` lives on `Logbook`, never on `LogbookTrip` (Pitfall 1). Do NOT create a migration — this repo has no `prisma/migrations/` directory (demo mode is the runtime default; DB users apply schema via `prisma migrate dev` themselves). Run `npx prisma validate` and `npx prisma generate` only.

Create `src/modules/logbook/validation.ts` (Zod, imported as in `src/modules/individual-tax/validation.ts`):

- `vehicleDetailsSchema`: `make`, `model`, `registrationNumber` — trimmed non-empty strings; `costPrice` — finite number > 0; `acquisitionDate` — optional/nullable `YYYY-MM-DD` string (regex `^\d{4}-\d{2}-\d{2}$` plus a real-date refine via `Date.parse` round-trip).
- `tripInputSchema`: `date` — required YYYY-MM-DD valid date; `businessKm` — finite number > 0; `fromLocation`, `toLocation`, `reason` — trimmed non-empty strings (SARS compulsory fields); `odometerStart`, `odometerEnd` — OPTIONAL nullable numbers >= 0; `.refine` that when BOTH odometer values are present, `odometerEnd >= odometerStart` AND businessKm is <= (odometerEnd - odometerStart) is NOT enforced at schema level (a trip may include private km) — only the end>=start ordering is enforced here.
- `logbookCreateSchema`: `clientId` non-empty string; `assessmentYear` — `z.number().int().min(2025).max(2027)` (2024 excluded per REQUIREMENTS.md scope decision); `vehicle` — `vehicleDetailsSchema` minus `id` (use `.omit({ id: true })` on a base object or a separate input schema); `openingOdometer` — finite number >= 0; `closingOdometer` — optional nullable number, with `.refine` closing >= opening when present.
- `actualExpensesSchema`: all five categories (`fuel`, `maintenance`, `insurance`, `licence`, `financeCharges`) — finite numbers >= 0, ALL required (a complete actual-expenses set; partial capture is rejected so `calculateActualCost` never runs on half data).
- `updateOdometersSchema`: `{ openingOdometer, closingOdometer }` with the same closing >= opening refine.
- `validateOdometerContinuity(input: { openingOdometer: number; closingOdometer: number | null; trips: Array<{ date: string; businessKm: number; odometerStart?: number | null; odometerEnd?: number | null }> }): { errors: LogbookWarning[]; warnings: LogbookWarning[] }` — a PURE exported function (Phase 4's import pipeline will reuse it — build once here, per Pitfall 2):
  - error `BUSINESS_KM_EXCEEDS_TOTAL` when closingOdometer is set and `sum(businessKm) > closingOdometer - openingOdometer` (compare with a 0.5 km tolerance to avoid float noise);
  - error `TRIP_ODOMETER_REVERSED` for any trip with both readings where end < start;
  - warning `TRIP_ODOMETER_DISCONTINUITY` when trips that have odometer readings, sorted by date (then by odometerStart), are not non-decreasing across trips (previous trip's end > next trip's start);
  - warning `CLOSING_ODOMETER_MISSING` when closingOdometer is null and at least one trip exists.

Create `src/modules/logbook/validation.test.ts` (Vitest, co-located per convention) covering at minimum:
1. trip without odometer readings PASSES (SARS optional-field rule — the load-bearing acceptance test);
2. trip missing each of date/businessKm/fromLocation/toLocation/reason FAILS (use `safeParse`, assert `success === false` per missing field);
3. trip with `odometerEnd < odometerStart` FAILS;
4. logbook with `closingOdometer < openingOdometer` FAILS; `assessmentYear` 2024 and 2028 FAIL; 2025/2026/2027 PASS;
5. `actualExpensesSchema` rejects a partial set (missing `financeCharges`);
6. `validateOdometerContinuity`: business-km-exceeds-total case returns the `BUSINESS_KM_EXCEEDS_TOTAL` error; reversed trip odometer returns `TRIP_ODOMETER_REVERSED`; out-of-order odometer sequence across two dated trips returns `TRIP_ODOMETER_DISCONTINUITY`; a clean logbook returns zero errors/warnings.
